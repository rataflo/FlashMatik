How to update Arduino firmware from Raspberry pi.

// 1: Copy file firmware.hex to /home/minutepapillons/photomaiton/Arduino
// 2: Launch commands.
//  If avrdude commande not recognized:
// 	sudo apt update
//	sudo apt install avrdude

sudo pkill -f python

cd /home/minutepapillons/photomaiton/arduino

avrdude -v -p atmega2560 -C /etc/avrdude.conf -c wiring -b 115200 -D -P /dev/flashmatik -U flash:w:firmware.hex:i
